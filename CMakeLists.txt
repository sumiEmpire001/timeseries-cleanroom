cmake_minimum_required(VERSION 3.12)
project(lord_sumi_cpp LANGUAGES CXX)
enable_testing()
add_executable(core_tests(cpp/tests.cpp)
target_include_directories(core_tests PRIVATE cpp)
set_target_properties(core_tests PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME core_tests COMMAND core_tests)
